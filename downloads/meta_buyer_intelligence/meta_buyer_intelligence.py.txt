def q(sql, a=(), one=False, commit=False):
    with L:
        c = sqlite3.connect(DB, timeout=30); c.row_factory = sqlite3.Row
        try:
            c.execute("PRAGMA synchronous=NORMAL")
            cur = c.execute(sql, a)

            r = [dict(x) for x in cur.fetchall()]; return (r[0] if r else None) if one else r
        finally: c.close()

# ---------------- PII (stdlib stream cipher; production: เปลี่ยนเป็น AES-GCM/KMS) ----------------
def _ks(n, nonce):
    o = b""; i = 0
    while len(o) < n: o += hmac.new(SECRET, nonce + i.to_bytes(4, "big"), "sha256").digest(); i += 1
    return o[:n]
def enc(t):
    b = (t or "").encode(); n = secrets.token_bytes(12); return n + bytes(x ^ y for x, y in zip(b, _ks(len(b), n)))

def create_order(d, actor="api"):

    pkg = next((p for p in S["retail_packages"] if p["dozen"] == dz), None)
    if not pkg: raise ValueError("ไม่มีแพ็กนี้ใน store config")
    ext = d.get("external_order_id")
    if ext:
        e = q("select id from orders where external_order_id=?", (ext,), one=True)
        if e: return e["id"]  # idempotent
    ck = None; c = d.get("customer")
    if c:
        ck = hashlib.sha256((str(c.get("phone", "")) + SECRET.hex()).encode()).hexdigest()[:16]
        q("insert or replace into customer_private values(?,?,?,?,?)", (TENANT, ck, enc(c.get("name")), enc(c.get("phone")), enc(c.get("address"))), commit=True)
    rev = d.get("gross_revenue", pkg["prepaid_total"] if pm == "PREPAID" else pkg["cod_total"])

       dz, pkg["units"], pm, rev, d.get("shipping_fee_charged_customer", 0), d.get("shipping_cost", S["shipping_cost"]), d.get("cod_fee", S["cod_fee"] if pm == "COD" else 0),

def patch_order(oid, d):
    o = q("select * from orders where id=?", (oid,), one=True)
    if not o: return None
    if "status" in d and d["status"] not in ST: raise ValueError("status ไม่ถูกต้อง")
    f = [k for k in ("status", "province", "region", "payment_method", "gross_revenue", "shipping_cost", "cod_fee", "ad_id", "campaign_id", "adset_id") if k in d]
    if f: q(f"update orders set {','.join(k+'=?' for k in f)},updated_at=? where id=?", [d[k] for k in f] + [now(), oid], commit=True)
    audit("api", "order.patch", f"order:{oid}", {k: o[k] for k in f}, {k: d[k] for k in f}); return q("select * from orders where id=?", (oid,), one=True)

def rng(days): t = today(); return str(t - timedelta(days=days - 1)), str(t)
def orders_between(a, b): return q("select * from orders where tenant_id=? and substr(created_at,1,10) between ? and ?", (TENANT, a, b))
def agg_orders(rows):
    r = dict(orders=0, revenue=0.0, cost=0.0, paid=0, repeat=0, cod=0, refused=0)
    for o in rows:
        if o["status"] == "CANCELLED": continue
        r["orders"] += 1; r["cost"] += (o["product_cost"] or 0) + (o["shipping_cost"] or 0) + (o["cod_fee"] or 0) + (o["packaging_cost"] or 0)
        if o["status"] in REV_OK: r["revenue"] += o["gross_revenue"] or 0
        r["paid"] += o["status"] in PAID_OK; r["repeat"] += not o["first_order"]
        if o["payment_method"] == "COD": r["cod"] += 1; r["refused"] += o["status"] == "REFUSED"
    return r
def base_tot(a, b):
    return q("select coalesce(sum(spend),0) spend,coalesce(sum(impressions),0) impressions,coalesce(sum(link_clicks),0) link_clicks,coalesce(sum(messages),0) messages from ad_insights where grain='base' and tenant_id=? and date between ? and ?", (TENANT, a, b), one=True)
def sd(x, y): return x / y if y else None

def wins(v):
    s = sorted(v); n = len(s); lo, hi = s[int(.05 * (n - 1))], s[min(n - 1, int(.95 * (n - 1) + .5))]; return [min(max(x, lo), hi) for x in v]
def norm(v):
    w = wins(v); lo, hi = min(w), max(w); return [0.5 if hi == lo else (x - lo) / (hi - lo) for x in w]

def segments(dim="ad", days=30):
    """Segment ที่มี order จริงให้ join ได้: ad / region / hour (age/placement ไม่มี order mapping จึงไม่คำนวณ score)"""
    a, b = rng(days); S = store(); g = {"ad": "base", "region": "region", "hour": "hourly"}[dim]
    ins = q("select seg,sum(spend) spend,sum(messages) msgs from ad_insights where grain=? and tenant_id=? and date between ? and ? group by seg", (g, TENANT, a, b))
    key = {"ad": lambda o: o["ad_id"], "region": lambda o: o["region"], "hour": lambda o: str(int(o["created_at"][11:13]))}[dim]
    grp = {}
    for o in orders_between(a, b): grp.setdefault(key(o), []).append(o)
    rows = []
    for r in ins:
        o = agg_orders(grp.get(r["seg"], [])); rows.append({"segment": r["seg"], "spend": r["spend"], "messages": r["msgs"], **o})
    if not rows: return []
    tm = sum(r["messages"] for r in rows); to = sum(r["orders"] for r in rows); ts = sum(r["spend"] for r in rows); gc = sd(to, tm) or 0; k = S["prior_strength"]
    for r in rows:
        r["close_rate"] = sd(r["orders"], r["messages"]); r["_close"] = (r["orders"] + k * gc) / (r["messages"] + k)
        r["cpa"] = sd(r["spend"], r["orders"]); r["roas"] = sd(r["revenue"], r["spend"]); r["contribution"] = r["revenue"] - r["cost"] - r["spend"]
        r["paid_rate"] = sd(r["paid"], r["orders"]); r["cod_refusal_rate"] = sd(r["refused"], r["cod"])
        r["confidence"] = round(0.4 * min(r["messages"] / 30, 1) + 0.6 * min(r["orders"] / 10, 1), 3)
    comp = [norm([r["_close"] for r in rows]), norm([1 / r["cpa"] if r["cpa"] else 0 for r in rows]),
            norm([r["contribution"] / r["orders"] if r["orders"] else min((x["contribution"] / x["orders"] for x in rows if x["orders"]), default=0) for r in rows]),
            norm([r["paid_rate"] or 0 for r in rows]), norm([sd(r["repeat"], r["orders"]) or 0 for r in rows]), norm([1 - (r["cod_refusal_rate"] or 0) for r in rows])]
    W = (.30, .25, .20, .10, .10, .05); gcpa = sd(ts, to)
    for i, r in enumerate(rows):
        r["buyer_score"] = round(100 * sum(w * c[i] for w, c in zip(W, comp)), 1); why = []
        if r["close_rate"] is not None and r["close_rate"] > gc: why.append("close rate สูงกว่าค่าเฉลี่ย")
        if r["cpa"] and gcpa and r["cpa"] < gcpa: why.append("cost/order ต่ำกว่าค่าเฉลี่ย")
        if r["orders"] == 0 and r["spend"] >= S["meaningful_spend"]: why.append("ใช้งบแล้วไม่มี order")
        if (r["cod_refusal_rate"] or 0) > S["cod_refusal_threshold"]: why.append("COD refusal สูง")
        r["why"] = why; sc, cf, m = r["buyer_score"], r["confidence"], r["spend"] >= S["meaningful_spend"]
        r["action"] = ("SCALE_CANDIDATE" if sc >= 75 and cf >= .65 else "REDUCE_CANDIDATE" if sc < 35 and m else "WATCH" if sc < 55 else "MAINTAIN_OR_TEST")
        if r["messages"] < S["min_messages"] or r["orders"] < S["min_orders"]: r["action"] = "LOW_SAMPLE"
        r.pop("_close")
    return sorted(rows, key=lambda r: (-r["confidence"] >= -0.0, -r["buyer_score"]))

            r["share"] = sd(r["spend"], ts) or 0; r["cpmsg"] = sd(r["spend"], r["msgs"]); r["ctr"] = sd(r["lc"], r["imp"]); idx = sd(r["cpmsg"], avg) if r["cpmsg"] and avg else None
            if r["msgs"] < S["min_messages"] or not idx: r["verdict"] = "ข้อมูลน้อย"
            else: r["verdict"] = "คุ้ม" if idx <= .8 else "แพงกว่าเฉลี่ย" if idx >= 1.3 else "ปกติ"
        out[g] = rows
    return out
def reconcile(days=30):

    return {"base_spend": base["spend"], "checks": out}

# ---------------- Agency dashboard extension (existing tokens unchanged) ----------------
DASH_AD_FIELDS = ('spend', 'impressions', 'clicks', 'link_clicks', 'messages', 'meta_purchases', 'meta_purchase_value')
DASH_ORDER_FIELDS = ('orders', 'revenue', 'order_cost', 'paid_orders', 'cod_orders', 'cod_refused', 'address_orders', 'address_revenue', 'unpriced_orders', 'paid_revenue', 'cod_settled', 'missing_cost_orders')

def dashboard_range(qs):
    days = int(qs.get('days', 30))
    if not 1 <= days <= 366: raise ValueError('days ต้องอยู่ระหว่าง 1–366')
    a, b = rng(days)
    a, b = qs.get('since', a), qs.get('until', b)
    start, end = datetime.strptime(a, '%Y-%m-%d').date(), datetime.strptime(b, '%Y-%m-%d').date()
    if start > end or (end-start).days > 365: raise ValueError('เลือกวันที่เรียงลำดับและไม่เกิน 366 วัน')
    return str(start), str(end)

def dashboard_derived(r):
    r.update(ctr=sd(r['clicks']*100, r['impressions']), link_ctr=sd(r['link_clicks']*100, r['impressions']),
             cpc=sd(r['spend'], r['clicks']), cpm=sd(r['spend']*1000, r['impressions']),
             cost_per_message=sd(r['spend'], r['messages']), cpa=sd(r['spend'], r['orders']),
             roas=sd(r['revenue'], r['spend']), close_rate=sd(r['orders']*100, r['messages']),
             contribution=r['revenue']-r['order_cost']-r['spend'],
             cod_refusal_rate=sd(r['cod_refused']*100, r['cod_orders']))
    return r

def dashboard_series(a, b, campaign=''):
    start, end = datetime.strptime(a, '%Y-%m-%d').date(), datetime.strptime(b, '%Y-%m-%d').date()
    daily = {}
    for i in range((end-start).days+1):
        d = str(start+timedelta(days=i))
        daily[d] = dict(date=d, has_ads=False, **dict.fromkeys(DASH_AD_FIELDS+DASH_ORDER_FIELDS, 0))
    where = 'tenant_id=? AND date BETWEEN ? AND ? AND grain=\'base\''
    args = [TENANT, a, b]
    if campaign: where += ' AND campaign_id=?'; args.append(campaign)
    insights = q('SELECT * FROM ad_insights WHERE '+where, args)
    for item in insights:
        r = daily[item['date']]; r['has_ads'] = True
        for field in DASH_AD_FIELDS: r[field] += item.get(field) or 0
    order_where = 'tenant_id=? AND substr(created_at,1,10) BETWEEN ? AND ?'
    order_args = [TENANT, a, b]
    if campaign: order_where += ' AND campaign_id=?'; order_args.append(campaign)
    orders = q('SELECT * FROM orders WHERE '+order_where, order_args)
    # This new dashboard separates confirmed sales from NEW/pending orders.
    successful = ('CONFIRMED', 'PAID', 'SHIPPED', 'DELIVERED', 'COD_COLLECTED')
    for item in orders:
        r = daily[item['created_at'][:10]]
        if item['status'] in successful:
            r['orders'] += 1; r['revenue'] += item['gross_revenue'] or 0
            r['order_cost'] += sum(item[k] or 0 for k in ('shipping_cost', 'cod_fee', 'product_cost', 'packaging_cost'))
        elif item['status'] in ('REFUSED', 'RETURNED'):
            r['order_cost'] += sum(item[k] or 0 for k in ('shipping_cost', 'cod_fee', 'packaging_cost'))
        if str(item.get('external_order_id') or '').startswith('chat:'):
            r['address_orders'] += 1
            r['address_revenue'] += item['gross_revenue'] or 0
        if item['status'] in successful and item['gross_revenue'] is None:r['unpriced_orders'] += 1
        if item['status'] in successful and any(item[k] is None for k in ('shipping_cost','cod_fee','product_cost','packaging_cost')):r['missing_cost_orders'] += 1
        paid=item['status'] in ('PAID','COD_COLLECTED') or (item['payment_method']=='PREPAID' and item['status']=='DELIVERED')
        r['paid_orders'] += int(paid)
        if paid:r['paid_revenue'] += item['gross_revenue'] or 0
        if item['payment_method'] == 'COD' and item['status'] != 'CANCELLED':
            r['cod_orders'] += 1; r['cod_refused'] += int(item['status'] == 'REFUSED')
    settled_where='tenant_id=? AND substr(cod_settled_at,1,10) BETWEEN ? AND ?'
    settled_args=[TENANT,a,b]
    if campaign:settled_where+=' AND campaign_id=?';settled_args.append(campaign)
    for settlement in q('SELECT cod_settled_at,cod_settled_amount FROM orders WHERE '+settled_where,settled_args):
        daily[settlement['cod_settled_at'][:10]]['cod_settled'] += settlement['cod_settled_amount'] or 0
    totals = dashboard_derived({k:sum(row[k] for row in daily.values()) for k in DASH_AD_FIELDS+DASH_ORDER_FIELDS})
    return {'daily':[dashboard_derived(row) for row in daily.values()], 'totals':totals,
            'ad_days':sum(row['has_ads'] for row in daily.values()), 'order_rows':len(orders)}

def agency_dashboard(qs):
    a, b = dashboard_range(qs); campaign = qs.get('campaign', '')
    current = dashboard_series(a, b, campaign)
    start = datetime.strptime(a, '%Y-%m-%d').date(); end = datetime.strptime(b, '%Y-%m-%d').date()
    duration = (end-start).days+1
    pa, pb = str(start-timedelta(days=duration)), str(start-timedelta(days=1))
    previous = dashboard_series(pa, pb, campaign)
    compare = {k:{'cur':v, 'prev':previous['totals'][k],
                  'delta':sd((v-previous['totals'][k])*100, abs(previous['totals'][k]))
                         if v is not None and previous['totals'][k] is not None else None}
               for k,v in current['totals'].items()}
    campaigns = q("SELECT campaign_id,max(campaign_name) name FROM ad_insights WHERE tenant_id=? AND grain='base' AND date BETWEEN ? AND ? AND campaign_id IS NOT NULL GROUP BY campaign_id ORDER BY name", (TENANT,a,b))
    dimension = qs.get('dimension', 'campaign')
    grains = {'campaign':('base','campaign_id'), 'ad':('base','ad_id'), 'age_gender':('age_gender','seg'),
              'region':('region','seg'), 'delivery':('delivery','seg'), 'device':('device','seg'), 'hourly':('hourly','seg')}
    if dimension not in grains: raise ValueError('dimension ไม่ถูกต้อง')
    grain, column = grains[dimension]
    where = 'tenant_id=? AND grain=? AND date BETWEEN ? AND ?'; args = [TENANT,grain,a,b]
    if campaign: where += ' AND campaign_id=?'; args.append(campaign)
    groups = q(f"SELECT {column} segment, max(ad_name) ad_name, max(campaign_name) campaign_name, sum(spend) spend, sum(impressions) impressions, sum(clicks) clicks, sum(link_clicks) link_clicks, sum(messages) messages FROM ad_insights WHERE {where} GROUP BY {column} ORDER BY spend DESC", args)
    for row in groups:
        row['label'] = (row['campaign_name'] if dimension=='campaign' else row['ad_name'] if dimension=='ad' else None) or row['segment'] or 'ไม่ระบุ'
        row['ctr'] = sd(row['clicks']*100,row['impressions']); row['cost_per_message'] = sd(row['spend'],row['messages'])
    warnings = list(JOB.get('warn') or [])
    if not current['ad_days']: warnings.append('ยังไม่มีข้อมูลแอดในช่วงนี้ — ช่องว่างกราฟไม่ใช่ยอดศูนย์ที่ยืนยันแล้ว')
    if not current['order_rows']: warnings.append('ยังไม่มีออเดอร์จริงในช่วงนี้ — ห้ามใช้ยอด Purchase ของ Meta แทนยอดขายจริง')
    if current['totals']['unpriced_orders']:warnings.append(f"มีออเดอร์ยืนยัน {current['totals']['unpriced_orders']} รายการยังไม่ทราบยอดเงิน — ยอดขายที่แสดงยังไม่ครบ กรุณาเติมยอดในเมนูรับออเดอร์จากแชต")
    if current['totals']['missing_cost_orders']:warnings.append(f"มี {current['totals']['missing_cost_orders']} ออเดอร์ยังขาดต้นทุน — ตัวเลขเหลือหลังต้นทุนยังไม่ครบ ให้เติมต้นทุนในรายละเอียดออเดอร์")
    if campaign: warnings.append('ยอดออเดอร์ในมุมมองแคมเปญนับเฉพาะออเดอร์ที่ระบุ campaign_id')
    return {**current, 'since':a, 'until':b, 'previous_since':pa, 'previous_until':pb,
            'previous_daily':previous['daily'], 'compare':compare, 'campaigns':campaigns,
            'groups':groups, 'dimension':dimension, 'currency':store().get('currency','THB'),
            'client_name':store().get('client_name',''), 'agency_name':store().get('agency_name',''),
            'mode':JOB['mode'], 'last_sync':JOB['last'], 'warnings':warnings}

RANK = {"INFO": 0, "WATCH": 1, "ACTION": 2, "URGENT": 3}
def push_out(sev, title, p):
    msg = f"[{sev}] {title}\n" + "\n".join(f"{k}: {v}" for k, v in p.items())
    print(msg)
    try:
        if env("TELEGRAM_BOT_TOKEN") and env("TELEGRAM_CHAT_ID"):
            urllib.request.urlopen(urllib.request.Request(f"https://api.telegram.org/bot{env('TELEGRAM_BOT_TOKEN')}/sendMessage",
                urllib.parse.urlencode({"chat_id": env("TELEGRAM_CHAT_ID"), "text": msg}).encode()), timeout=10)
        if env("ALERT_WEBHOOK_URL"):
            urllib.request.urlopen(urllib.request.Request(env("ALERT_WEBHOOK_URL"), json.dumps({"severity": sev, "title": title, "payload": p}).encode(), {"Content-Type": "application/json"}), timeout=10)
    except Exception as e: print("alert delivery failed:", type(e).__name__)

def alert(key, sev, title, p):
    last = q("select * from alert_events where tenant_id=? and alert_key=? order by id desc limit 1", (TENANT, key), one=True)
    if last:
        age = (datetime.now(TZ).replace(tzinfo=None) - datetime.strptime(last["created_at"], "%Y-%m-%d %H:%M:%S")).total_seconds() / 3600
        if age < store()["alert_cooldown_hours"] and RANK[sev] <= RANK[last["severity"]]: return
    q("insert into alert_events(tenant_id,created_at,alert_key,severity,title,payload) values(?,?,?,?,?,?)", (TENANT, now(), key, sev, title, json.dumps(p, ensure_ascii=False)), commit=True); push_out(sev, title, p)

def rec(key, sev, typ, entity, evidence, action):
    if not q("select 1 x from orders limit 1", one=True): return  # ไม่มีออเดอร์ = ไม่สรุปเรื่องยอดขาย
    if q("select id from recommendations where rec_key=? and status in ('PENDING','APPROVED')", (key,), one=True): return
    q("insert into recommendations(tenant_id,rec_key,created_at,severity,type,entity,evidence,recommendation) values(?,?,?,?,?,?,?,?)",
      (TENANT, key, now(), sev, typ, entity, json.dumps(evidence, ensure_ascii=False), json.dumps({"action": action, "requires_approval": True}, ensure_ascii=False)), commit=True)
    alert(key, sev, f"{typ}: {entity}", {"เกิดอะไรขึ้น": evidence, "แนะนำ": action, "ต้องอนุมัติ": "ใช่ (ไม่มีการแก้ Meta จนกว่าจะอนุมัติ)"})

CHAT_SCHEMA = """
CREATE TABLE IF NOT EXISTS chat_conversations(tenant_id TEXT,channel TEXT,conversation_key TEXT,context_enc BLOB,order_key TEXT,order_id INTEGER,updated_at TEXT,PRIMARY KEY(tenant_id,channel,conversation_key));

def init_chat_schema(connection):
    connection.executescript(CHAT_SCHEMA)
    present = {row[1] for row in connection.execute('PRAGMA table_info(orders)')}
    for key,kind in {'tracking_number':'TEXT','carrier':'TEXT','last_shipping_at':'TEXT','cod_settled_amount':'REAL DEFAULT 0','cod_settled_at':'TEXT','amount_source':'TEXT'}.items():
        if key not in present: connection.execute(f'ALTER TABLE orders ADD COLUMN {key} {kind}')
    connection.execute('CREATE INDEX IF NOT EXISTS ix_orders_tracking ON orders(tenant_id,carrier,tracking_number)')

def chat_key(value): return hmac.new(SECRET,str(value).encode(),'sha256').hexdigest()

def chat_seal(value):
    data=enc(json.dumps(value,ensure_ascii=False))
    return hmac.new(SECRET,b'chat-context:'+data,'sha256').digest()+data

def chat_open(value):
    if not value: return {}
    tag,data=value[:32],value[32:]
    if not hmac.compare_digest(tag,hmac.new(SECRET,b'chat-context:'+data,'sha256').digest()): raise ValueError('ข้อมูลแชตเสียหายหรือ APP_SECRET ไม่ตรงกับเดิม')
    nonce,content=data[:12],data[12:]
    return json.loads(bytes(a^b for a,b in zip(content,_ks(len(content),nonce))).decode())

def chat_time(value=None):
    if isinstance(value,(int,float)):
        stamp=float(value)/1000 if float(value)>100000000000 else float(value)
        result=datetime.fromtimestamp(stamp,TZ)
    elif value:
        result=datetime.fromisoformat(str(value).replace('Z','+00:00'))
        if result.tzinfo is None: result=result.replace(tzinfo=TZ)
    else: result=datetime.now(TZ)
    return result.astimezone(TZ).isoformat(timespec='seconds')

def chat_parse(customer_text,quote_text='',fields=None,settings=None):
    customer_text=customer_text.translate(str.maketrans('๐๑๒๓๔๕๖๗๘๙','0123456789'))
    fields=dict(fields or {})
    settings=settings if settings is not None else store()
    text=customer_text+'\n'+quote_text
    text=text.translate(str.maketrans('๐๑๒๓๔๕๖๗๘๙','0123456789'))
    quantities=re.findall(r'(?:จำนวน\s*[:：]?\s*)?([1-9][0-9]*)\s*โหล',text)
    if quantities:
        qty=int(quantities[-1])
        if 1<=qty<=1000:fields['quantity_dozen']=qty
    payments=list(re.finditer(r'เก็บเงินปลายทาง|ปลายทาง|\bCOD\b|โอนเงิน|ชำระโอน|โอนแล้ว|โอนค่ะ|โอนครับ|\bPREPAID\b',text,re.I))
    if payments: fields['payment_method']='COD' if re.search(r'ปลายทาง|COD',payments[-1].group(),re.I) else 'PREPAID'
    totals=re.findall(r'(?:ยอดรวม(?:ทั้งหมด)?|รวมยอด|รวมเงิน|ยอดเงินรวม|ยอดชำระ|รวมทั้งสิ้น|รวม(?:ค่าส่ง)?(?:ทั้งหมด)?|ยอดทั้งหมด)\s*[:：=]?\s*(?:฿\s*)?([0-9][0-9,]*(?:\.[0-9]{1,2})?)(?:\s*(?:บาท|฿))?',text)
    primary_totals=re.findall(r'(?:ยอดรวม(?:ทั้งหมด)?|รวมยอด|รวมเงิน|ยอดเงินรวม|ยอดชำระ|รวมทั้งสิ้น|ยอดทั้งหมด)\s*[:：=]?\s*(?:฿\s*)?([0-9][0-9,]*(?:\.[0-9]{1,2})?)',text)
    if primary_totals:totals=primary_totals
    if totals:
        fields['gross_revenue']=money_number(totals[-1]);fields['amount_source']='chat_total'
    if fields.get('gross_revenue') is None and fields.get('quantity_dozen') and fields.get('payment_method'):
        package=next((p for p in settings['retail_packages'] if p['dozen']==fields['quantity_dozen']),None)
        if package:
            fields['gross_revenue']=float(package['cod_total' if fields['payment_method']=='COD' else 'prepaid_total']);fields['amount_source']='store_package'
    phones=re.findall(r'(?<!\d)(0[689](?:[ -]?\d){8})(?!\d)',customer_text)
    if phones: fields['phone']=re.sub(r'\D','',phones[-1])
    names=re.findall(r'(?:ชื่อผู้รับ|ผู้รับ|ชื่อ)\s*[:：]\s*([^\n]+)',customer_text)
    if names: fields['name']=names[-1].strip()
    zipcode=re.search(r'(?<!\d)[1-9]\d{4}(?!\d)',customer_text)
    house=re.search(r'(?:บ้านเลขที่\s*)?\b\d{1,5}/\d{1,5}\b|(?:บ้านเลขที่|เลขที่)\s*\d+',customer_text)
    area=re.search(r'ตำบล|แขวง|อำเภอ|เขต|จังหวัด|กรุงเทพ|\bต\.|\bอ\.|\bจ\.',customer_text)
    address_seen=bool((zipcode and area) or (zipcode and house) or (house and area and phones))
    if address_seen:
        lines=[line.strip() for line in customer_text.splitlines() if line.strip()]
        address_lines=[line for line in lines if re.search(r'ตำบล|แขวง|อำเภอ|เขต|จังหวัด|กรุงเทพ|บ้านเลขที่|เลขที่|\d+/\d+|(?<!\d)[1-9]\d{4}(?!\d)',line)]
        fields['address']='\n'.join(address_lines) or customer_text
    province=re.findall(r'(?:จังหวัด|จ\.)\s*([ก-๙]+)',customer_text)
    if province:fields['province']=province[-1]
    elif re.search(r'กรุงเทพ',customer_text):fields['province']='กรุงเทพมหานคร'
    fields['address_seen']=address_seen or bool(fields.get('address'))
    return fields

def money_number(value):
    import math
    try: number=float(str(value).replace(',',''))
    except (ValueError,TypeError): raise ValueError('ยอดเงินต้องเป็นตัวเลข') from None
    if not math.isfinite(number) or number<0: raise ValueError('ยอดเงินต้องไม่ติดลบและเป็นตัวเลขจำกัด')
    return number

def receive_chat(data):
    channel=str(data.get('channel','manual')).lower()
    if channel not in ('manual','messenger','line'): raise ValueError('channel ต้องเป็น manual, messenger หรือ line')
    event_id=str(data.get('message_id') or '').strip()
    conversation=str(data.get('conversation_id') or '').strip()
    text=str(data.get('text') or '').strip()
    if not event_id or not conversation or not text or len(text)>20000: raise ValueError('ต้องมี message_id, conversation_id และ text ไม่เกิน 20000 ตัวอักษร')
    conversation_key=chat_key(channel+':'+conversation);event_id=chat_key(channel+':'+event_id)
    created_at=chat_time(data.get('created_at'))
    direction=data.get('direction','customer')
    if direction not in ('customer','shop'): raise ValueError('direction ต้องเป็น customer หรือ shop')
    settings=store()
    with L:
        connection=sqlite3.connect(DB,timeout=30);connection.row_factory=sqlite3.Row
        try:
            with connection:
                duplicate=connection.execute('SELECT order_id FROM chat_events WHERE tenant_id=? AND channel=? AND event_id=?',(TENANT,channel,event_id)).fetchone()
                if duplicate: return {'duplicate':True,'order_id':duplicate['order_id']}
                existing=connection.execute('SELECT * FROM chat_conversations WHERE tenant_id=? AND channel=? AND conversation_key=?',(TENANT,channel,conversation_key)).fetchone()
                context=chat_open(existing['context_enc']) if existing else {'customer':[],'shop':[],'fields':{}}
                order_id=existing['order_id'] if existing else None
                order_key=existing['order_key'] if existing else event_id
                # Repeated address corrections stay in the same order. New sales use an explicit order boundary.
                new_intent=direction=='customer' and bool(re.search(r'สั่งเพิ่ม|สั่งอีก|สั่งใหม่|ซื้ออีก|เอาอีก|ออเดอร์ใหม่',text))
                if data.get('new_order') is True or (new_intent and order_id is not None and created_at>=context.get('latest_at','')):
                    context={'customer':[],'shop':[],'fields':{}};order_id=None;order_key=event_id
                context['latest_at']=max(context.get('latest_at',''),created_at)
                context.setdefault(direction,[]).append(text)
                context[direction]=context[direction][-30:]
                fields=chat_parse('\n'.join(context['customer']),'\n'.join(context.get('shop',[])),context.get('fields'),settings)
                if direction=='customer':
                    current_fields=chat_parse(text,'',{},settings)
                    if current_fields.get('address_seen'):fields['address']=current_fields['address']
                context['fields']=fields
                context.setdefault('attribution',{}).update({key:data[key] for key in ('campaign_id','adset_id','ad_id') if data.get(key)})
                if fields.get('address_seen') and (direction=='customer' or order_id is not None):
                    pkg=next((p for p in settings['retail_packages'] if p['dozen']==fields.get('quantity_dozen')),None)
                    # No invented total/payment: unknown values remain NULL/UNKNOWN and are visible for review.
                    amount=fields.get('gross_revenue');payment=fields.get('payment_method','UNKNOWN');quantity=fields.get('quantity_dozen')
                    customer_key=chat_key(channel+':'+conversation)
                    connection.execute('INSERT OR REPLACE INTO customer_private VALUES(?,?,?,?,?)',(TENANT,customer_key,enc(fields.get('name')),enc(fields.get('phone')),enc(fields.get('address'))))
                    if order_id is None:
                        external='chat:'+channel+':'+conversation_key[:16]+':'+order_key[:16]
                        cursor=connection.execute('''INSERT INTO orders(tenant_id,external_order_id,created_at,customer_key,source_channel,campaign_id,adset_id,ad_id,conversation_id,quantity_dozen,quantity_units,payment_method,gross_revenue,shipping_fee_charged_customer,shipping_cost,cod_fee,product_cost,packaging_cost,status,province,region,first_order,updated_at,amount_source) VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)''',
                            (TENANT,external,created_at,customer_key,channel,context['attribution'].get('campaign_id'),context['attribution'].get('adset_id'),context['attribution'].get('ad_id'),conversation_key,quantity,pkg['units'] if pkg else (quantity*12 if quantity else None),payment,amount,0,settings['shipping_cost'],settings['cod_fee'] if payment=='COD' else 0,settings['product_cost_per_dozen']*quantity if quantity else None,settings['packaging_cost'],'CONFIRMED',fields.get('province'),fields.get('province'),1,now(),fields.get('amount_source')))
                        order_id=cursor.lastrowid
                        prior=connection.execute('SELECT 1 FROM orders WHERE tenant_id=? AND customer_key=? AND id<>? LIMIT 1',(TENANT,customer_key,order_id)).fetchone()
                        if prior:connection.execute('UPDATE orders SET first_order=0 WHERE id=?',(order_id,))
                        connection.execute('INSERT INTO audit_logs(tenant_id,created_at,actor,action,entity,after_json) VALUES(?,?,?,?,?,?)',(TENANT,now(),'chat','order.address_confirmed',f'order:{order_id}',json.dumps({'status':'CONFIRMED','amount_known':amount is not None})))
                    else:
                        order=connection.execute('SELECT * FROM orders WHERE tenant_id=? AND id=?',(TENANT,order_id)).fetchone()
                        # Do not alter a booked total from subsequent unstructured chat. Operator can edit it explicitly.
                        if order and order['gross_revenue'] is None and amount is not None:
                            connection.execute('UPDATE orders SET gross_revenue=?,amount_source=?,updated_at=? WHERE id=?',(amount,fields.get('amount_source'),now(),order_id))
                        if order and order['payment_method']=='UNKNOWN' and payment!='UNKNOWN':
                            connection.execute('UPDATE orders SET payment_method=?,updated_at=? WHERE id=?',(payment,now(),order_id))
                            if payment=='COD':connection.execute('UPDATE orders SET cod_fee=? WHERE id=?',(settings['cod_fee'],order_id))

                            connection.execute('UPDATE orders SET quantity_dozen=?,quantity_units=?,updated_at=? WHERE id=?',(quantity,pkg['units'] if pkg else quantity*12,now(),order_id))

                connection.execute('INSERT OR REPLACE INTO chat_conversations VALUES(?,?,?,?,?,?,?)',(TENANT,channel,conversation_key,chat_seal(context),order_key,order_id,now()))
                connection.execute('INSERT INTO chat_events VALUES(?,?,?,?,?,?)',(TENANT,channel,event_id,conversation_key,created_at,order_id))
                return {'duplicate':False,'order_id':order_id,'address_seen':fields.get('address_seen',False),'amount_known':fields.get('gross_revenue') is not None}

def review_chat_order(order_id,body):
    order=q('SELECT * FROM orders WHERE tenant_id=? AND id=?',(TENANT,order_id),one=True)

    patch={}

SHIPPING_STATUS={'SHIPPED':'SHIPPED','IN_TRANSIT':'SHIPPED','DELIVERED':'DELIVERED','REFUSED':'REFUSED','RETURNED':'RETURNED','COD_COLLECTED':'COD_COLLECTED'}

def attach_tracking(body):
    order_id=int(body['order_id']);tracking=str(body.get('tracking_number','')).strip()
    if not tracking or len(tracking)>100:raise ValueError('กรอกเลขพัสดุ')
    carrier=str(body.get('carrier','FLASH')).upper()

    with L:
        connection=sqlite3.connect(DB,timeout=30)

                other=connection.execute('SELECT id FROM orders WHERE tenant_id=? AND carrier=? AND tracking_number=? AND id<>?',(TENANT,carrier,tracking,order_id)).fetchone()
                if other:raise ValueError('เลขพัสดุผูกกับออเดอร์อื่นแล้ว')

def receive_shipping(body):
    event_id=str(body.get('event_id','')).strip();tracking=str(body.get('tracking_number','')).strip()
    raw_status=str(body.get('status','')).upper();carrier=str(body.get('carrier','FLASH')).upper()
    if not event_id or not tracking or not body.get('event_time') or carrier!='FLASH' or raw_status not in SHIPPING_STATUS:raise ValueError('ต้องมี event_id, tracking_number, carrier=FLASH และ status ตามคู่มือ')
    event_time=chat_time(body.get('event_time'));status=SHIPPING_STATUS[raw_status]
    amount=money_number(body['cod_amount']) if status=='COD_COLLECTED' and body.get('cod_amount') is not None else None
    if status=='COD_COLLECTED' and amount is None:raise ValueError('COD_COLLECTED ต้องระบุ cod_amount ที่ขนส่งโอนจริง')
    with L:
        connection=sqlite3.connect(DB,timeout=30);connection.row_factory=sqlite3.Row
        try:
            with connection:
                if connection.execute('SELECT 1 FROM shipping_events WHERE tenant_id=? AND event_id=?',(TENANT,event_id)).fetchone():return {'duplicate':True}
                order=connection.execute('SELECT * FROM orders WHERE tenant_id=? AND carrier=? AND tracking_number=?',(TENANT,carrier,tracking)).fetchone()
                if not order:raise ValueError('ยังไม่ผูกเลขพัสดุนี้กับออเดอร์')
                applied=not order['last_shipping_at'] or event_time>order['last_shipping_at']
                if order['status'] in ('COD_COLLECTED','CANCELLED','REFUSED','RETURNED') and status!=order['status']:applied=False
                if order['status']=='DELIVERED' and status=='SHIPPED':applied=False
                if status=='COD_COLLECTED' and order['payment_method']!='COD':raise ValueError('ออเดอร์นี้ไม่ใช่ COD')
                if applied:
                    connection.execute('UPDATE orders SET status=?,last_shipping_at=?,updated_at=? WHERE id=?',(status,event_time,now(),order['id']))
                    if status=='COD_COLLECTED':connection.execute('UPDATE orders SET cod_settled_amount=?,cod_settled_at=COALESCE(cod_settled_at,?) WHERE id=?',(amount,event_time,order['id']))
                connection.execute('INSERT INTO shipping_events VALUES(?,?,?,?,?,?,?,?,?)',(TENANT,event_id,order['id'],carrier,tracking,status,event_time,amount,int(applied)))
                return {'duplicate':False,'order_id':order['id'],'applied':bool(applied),'status':status}
        finally:connection.close()

def import_shipping_csv(text):
    import csv,io
    rows=list(csv.DictReader(io.StringIO(str(text).lstrip('\ufeff'))))
    if len(rows)>2000:raise ValueError('นำเข้าได้ครั้งละไม่เกิน 2000 แถว')
    outcomes=[]
    for number,row in enumerate(rows,2):
        if not row.get('event_id'):row['event_id']='csv:'+hashlib.sha256(json.dumps(row,sort_keys=True).encode()).hexdigest()
        try:outcomes.append({'line':number,**receive_shipping(row)})
        except (ValueError,KeyError):outcomes.append({'line':number,'error':'ตรวจเลขพัสดุ สถานะ วันเวลา และยอด COD ตามคู่มือ'})
    return {'rows':outcomes,'ok':sum(not row.get('error') for row in outcomes),'failed':sum(bool(row.get('error')) for row in outcomes)}

def native_chat_events(channel,payload):
    records=[]
    if channel=='messenger':
        if payload.get('object')!='page':raise ValueError('ต้องเป็น Messenger Page webhook')
        for entry in payload.get('entry',[]):
            for event in entry.get('messaging',[]):
                message=event.get('message',{})
                if not message.get('text') or not message.get('mid'):continue
                shop=bool(message.get('is_echo'))
                page_id=str(event.get('sender' if shop else 'recipient',{}).get('id',''))
                person=str(event.get('recipient' if shop else 'sender',{}).get('id',''))
                if not page_id or not person:continue
                referral=event.get('referral') or message.get('referral') or {}
                records.append({'channel':'messenger','message_id':message['mid'],'conversation_id':page_id+':'+person,
                                'text':message['text'],'created_at':event.get('timestamp'),'direction':'shop' if shop else 'customer',
                                'ad_id':referral.get('ad_id')})
    elif channel=='line':
        for event in payload.get('events',[]):
            message=event.get('message',{});source=event.get('source',{})
            if event.get('type')!='message' or message.get('type')!='text' or source.get('type')!='user':continue
            event_id=event.get('webhookEventId') or message.get('id');person=source.get('userId')
            if not event_id or not person:continue
            records.append({'channel':'line','message_id':event_id,'conversation_id':str(payload.get('destination',''))+':'+person,
                            'text':message['text'],'created_at':event.get('timestamp')})
    return [receive_chat(record) for record in records]

def handle_chat_webhook(handler,method,path,query):
    import base64
    if path not in ('/webhooks/messenger','/webhooks/line'):return False
    channel=path.rsplit('/',1)[-1]
    if method=='GET' and channel=='messenger':
        expected=env('WEBHOOK_VERIFY_TOKEN')
        if not expected or query.get('hub.mode')!='subscribe' or not hmac.compare_digest(query.get('hub.verify_token',''),expected):handler.out(403,{'error':'verification failed'});return True
        handler.out(200,str(query.get('hub.challenge','')).encode(),'text/plain');return True
    if method!='POST':handler.out(405,{'error':'method not allowed'});return True
    secret=env('META_APP_SECRET' if channel=='messenger' else 'LINE_CHANNEL_SECRET')
    if not secret:handler.out(503,{'error':'webhook secret not configured'});return True
    try:
        size=int(handler.headers.get('Content-Length') or 0)
        if size<=0 or size>1048576:raise ValueError('payload size')
        raw=handler.rfile.read(size)
        digest=hmac.new(secret.encode(),raw,'sha256').digest()
        signature=handler.headers.get('X-Hub-Signature-256' if channel=='messenger' else 'X-Line-Signature','')
        expected='sha256='+digest.hex() if channel=='messenger' else base64.b64encode(digest).decode()
        if not hmac.compare_digest(signature,expected):handler.out(403,{'error':'invalid signature'});return True
        results=native_chat_events(channel,json.loads(raw))
        handler.out(200,{'received':len(results)})
    except (ValueError,TypeError,KeyError):handler.out(400,{'error':'invalid webhook payload'})
    except Exception:handler.out(500,{'error':'webhook processing failed'})
    return True

def extend_token():
    """แลกโทเคนอายุสั้นเป็นอายุ ~60 วัน แล้วเขียนทับ META_ACCESS_TOKEN ใน config.env/.env"""
    tok, aid, sec = env("META_ACCESS_TOKEN"), env("META_APP_ID", "1274393901342976"), env("META_APP_SECRET")
    if not tok: print("ไม่พบ META_ACCESS_TOKEN ใน config.env - วางโทเคนก่อน"); return
    if not sec:
        import getpass
        sec = getpass.getpass("Paste App Secret then press Enter (hidden, nothing will show): ").strip()
    if not sec: print("ไม่ได้ใส่ App Secret"); return
    url = f"https://graph.facebook.com/{env('META_GRAPH_API_VERSION', 'v21.0')}/oauth/access_token?" + urllib.parse.urlencode(
        {"grant_type": "fb_exchange_token", "client_id": aid, "client_secret": sec, "fb_exchange_token": tok})
    try: d = json.load(urllib.request.urlopen(url, timeout=30))
    except urllib.error.HTTPError as e:
        try: msg = json.load(e).get("error", {}).get("message", "")
        except Exception: msg = ""
        print("ต่ออายุไม่สำเร็จ:", msg or e.code, "\n(ถ้าโทเคนหมดอายุแล้ว ให้ Generate ใหม่ใน Graph API Explorer แล้วลองอีกครั้ง)"); return
    except Exception as e: print("ต่ออายุไม่สำเร็จ:", type(e).__name__); return
    new = d.get("access_token")
    if not new: print("Meta ไม่ส่งโทเคนกลับมา"); return
    done = False
    for f in (".env", "config.env"):
        path = os.path.join(ROOT, f)
        if os.path.exists(path):
            txt = open(path, encoding="utf-8-sig").read()
            if re.search(r"(?m)^META_ACCESS_TOKEN=", txt):
                open(path, "w", encoding="utf-8").write(re.sub(r"(?m)^META_ACCESS_TOKEN=.*$", lambda m: "META_ACCESS_TOKEN=" + new, txt)); done = True; break
    print(f"สำเร็จ! โทเคนใหม่อายุประมาณ {int(d.get('expires_in', 0)) // 86400} วัน", "- บันทึกลง config.env แล้ว ปิดและเปิดโปรแกรมใหม่" if done else "- แต่หาไฟล์ config.env ไม่เจอ จึงยังไม่ได้บันทึก")
